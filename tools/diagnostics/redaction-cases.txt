# Header lines set what the shell knows when it builds a report:
#   @name    a character name (from tokens.json and the engine) -> <player>
#   @secret  a literal value from tokens.json (session, character id, tokens) -> <redacted>
# Then pairs: "< input line" and "> the line after redaction". A line with nothing to remove comes back unchanged.

@name Quillfeather
@secret 7f3c2a91d0e4b5a6
